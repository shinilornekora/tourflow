-- 3. остатки равны числу подтверждённых броней на каждые сутки (фоновая сверка)
WITH actual AS (
  SELECT r.room_type_id, d::date AS day, count(*)::int AS confirmed
  FROM reservations r, generate_series(r.check_in, r.check_out - 1, interval '1 day') d
  WHERE r.status = 'confirmed' GROUP BY 1, 2)
SELECT count(*) AS violations FROM inventory i FULL JOIN actual a ON a.room_type_id = i.room_type_id AND a.day = i.day
WHERE coalesce(i.booked, 0) <> coalesce(a.confirmed, 0);
 violations 
------------
          0

-- 4. один ключ идемпотентности — одна бронь, и эта бронь существует
SELECT (SELECT count(*) FROM (SELECT reservation_id FROM idempotency_keys GROUP BY 1 HAVING count(*) > 1) x) AS duplicate_keys,
       (SELECT count(*) FROM idempotency_keys k LEFT JOIN reservations r ON r.id = k.reservation_id WHERE r.id IS NULL) AS keys_without_reservation;

-- 5. у каждого ключа есть сохранённый ответ и запись аудита о создании
SELECT count(*) AS violations FROM idempotency_keys k
  LEFT JOIN audit_log a ON a.entity_id = k.reservation_id AND a.action = 'reservation.created'
  WHERE k.response IS NULL OR a.id IS NULL;

-- 5а. обратная сторона: у каждой записи аудита о брони есть бронь (журнал не ссылается на удалённое)
SELECT count(*) AS violations FROM audit_log a LEFT JOIN reservations r ON r.id = a.entity_id
